def check_index(index, lower_bound, upper_bound):
inBounds = True
if index < lower_bound:
index = lower_bound
inBounds = False
elif index >= upper_bound:
index = upper_bound - 1
inBounds = False

return index, inBounds

def show_flag_tf_scenechanged(self, img, idx_thisImg):
    img = cv2.resize(img, (1750, 900), interpolation=cv2.INTER_LINEAR)
    font = cv2.FONT_HERSHEY_SIMPLEX


    reloc_flag = self.jdata["pose_info"][idx_thisImg]["reloc"] # 当前帧的定位结果(那个大的Ture&False)
    font_color = (0, 255, 0) if reloc_flag else (0, 0, 255) #  green for True, red for False

    # Define the position for the text and rectangle
    text_position = (450, 400)

    # Get the text size
    (text_width, text_height), baseline = cv2.getTextSize(str(reloc_flag), font, 3, 4)

    # Calculate the rectangle position based on the text position
    rect_top_left = (text_position[0], text_position[1] - text_height - baseline)
    rect_bottom_right = (text_position[0] + text_width, text_position[1] + baseline)

    # Draw a white rectangle as background
    cv2.rectangle(img, rect_top_left, rect_bottom_right, (255, 255, 255), thickness=cv2.FILLED)

    # Draw the text on top of the rectangle
    cv2.putText(img, str(reloc_flag), text_position, font, 3, font_color, 4) # 在图像上显示reloc_flag

def save_manually(self):
    # 确保 jsonsDir_filtered 存在。如不存在，则创建它。
    os.makedirs(self.jsonsDir_filtered, exist_ok=True)
    bak_folder = os.path.join(self.jsonsDir_filtered, "bak")
    os.makedirs(bak_folder, exist_ok=True)  # 创建备份文件夹
    # 拼凑出保存当前clip的名字
    savePath_currClip = os.path.join(self.jsonsDir_filtered, self.fname_currClip)
    # 拼凑出保存在bak文件夹中的clip的名字
    def copy_and_rename_json(source_path, destination_dir):
        # 确保源文件存在
        if not os.path.exists(source_path):
            # print(f"源文件 {source_path} 不存在！")
            return

        # 创建目标目录（如果不存在）
        os.makedirs(destination_dir, exist_ok=True)

        # 获取文件名和扩展名
        source_filename = os.path.basename(source_path)
        filename, ext = os.path.splitext(source_filename)

        # 目标目录中已存在的文件列表
        existing_files = os.listdir(destination_dir)
        version_pattern = re.compile(rf"{filename}_v(\d+)\.json")

        # 查找已存在的版本号
        existing_versions = []
        for file in existing_files:
            match = version_pattern.match(file)
            if match:
                existing_versions.append(int(match.group(1)))

        # 确定新的版本号
        if not existing_versions:
            new_version = 0
        else:
            new_version = max(existing_versions) + 1

        # 构建新的文件名和目标路径
        new_filename = f"{filename}_v{new_version}{ext}"
        destination_path = os.path.join(destination_dir, new_filename)

        # 拷贝文件
        shutil.copy2(source_path, destination_path)
        # print(f"文件已拷贝到 {destination_path}")
    copy_and_rename_json(savePath_currClip,bak_folder)
    
    # 更新jdata
    self.jdata["pcd_list"] = self.newPcdList
    self.jdata["pose_info"] = self.newPoseInfo
    self.jdata["cnt_reloc_vaild"] = self.count_valid()
    # print(f"self.idxFrame 为{self.idxFrame}")
    self.jdata["idxFrame_leftLastTime"] = self.idxFrame # add/update this pair
    self.jdata["frame_num"] = len(self.newPcdList) # add/update this pair
    self.jdata["time"] = self.time
    # 排序 JSON 数据
    sorted_dict = {
        key: self.jdata[key]
        for key in sorted(self.jdata.keys())
    }

    json.dump(
        sorted_dict,
        open(savePath_currClip, "w"),
        # indent=4, # 格式化输出，缩进 4 个空格，便于阅读。
        ensure_ascii=False, # 允许非 ASCII 字符（如中文）正常保存。
    )
    # self.warning = f"saved, {savePath_currClip}"
    self.warning = f"saved"
    print(self.warning)


def set_following_xframes_reloc_to_value(self, value_toSet, x_frames):
    for i in range(x_frames):
        if self.idxFrame + i < len(self.newPcdList):
            idx_thisImg = self.newPcdList[self.idxFrame + i]
            self.jdata["pose_info"][idx_thisImg]["reloc"] = value_toSet
    self.warning = f"set last {x_frames} {value_toSet}"

def input_helper(self, step, direction):
    if direction == "backward":
        self.idxFrame -= step

        self.idxFrame, inBounds = check_index(
            index=self.idxFrame,
            lower_bound=0,
            upper_bound=len(self.newPcdList)
        )
        if not inBounds:
            self.warning = f"idxFrame out of bounds"
        else:
            self.warning = f"{direction}ed {step} frames"
    elif direction == "forward":
        self.idxFrame += step

        self.idxFrame, inBounds = check_index(
            index=self.idxFrame,
            lower_bound=0,
            upper_bound=len(self.newPcdList),
        )
        if not inBounds:
            self.warning = f"idxFrame out of bounds"
        else:
            self.warning = f"{direction}ed {step} frames"

def input(self, key):
    # backward
    if key == ord(key_prev_frame_2):
        self.input_helper(step=20, direction="backward")
    # forward
    elif key == ord(key_next_frame_2):
        self.input_helper(step=20, direction="forward")

    elif key == ord(key_set_true_next_1_frame):
        step = 20 # 再只用改这个值就行了
        self.set_following_xframes_reloc_to_value(value_toSet=True, x_frames=step)
        self.warning = f"set last {step} True"
        self.idxFrame += step
    elif key == ord(key_set_false_next_1_frame):
        step = 20 # 再只用改这个值就行了
        self.set_following_xframes_reloc_to_value(value_toSet=False, x_frames=step)
        self.warning = f"set last {step} False"
        self.idxFrame += step
    elif key == ord(autokey_set_false_next1):
        step = 1
        self.set_following_xframes_reloc_to_value(value_toSet=False, x_frames=step)
        self.warning = f"prev imgs are none, auto set to F"
        self.idxFrame += step

    elif key == ord(key_set_curr_and_next_all_true):
        self.set_following_xframes_reloc_to_value(value_toSet=True, x_frames=len(self.newPcdList) - self.idxFrame)

        self.save_manually()
